7. void sort_journal_by_average(Student journal[], int count); 
    Сортировка журнала "Ранжировать учеников в классе по успеваемости... у кого средний бал выше - тот находится в верху списка вывода".

    Логика в task_1.c:
    /* Функция-компаратор для qsort. Сортирует по убыванию среднего балла */
    static int compare_students(const void *a, const void *b) {
        /* Приводим void* к Student* */
        const Student *s1 = (const Student *)a;
        const Student *s2 = (const Student *)b;
        /* Получаем средние баллы */
        float avg1 = get_student_average(s1);
        float avg2 = get_student_average(s2);
        /* 
        * qsort ожидает:
        * < 0, если s1 должно быть выше s2
        * > 0, если s1 должно быть ниже s2
        * = 0, если они равны
        * Нам нужно по УБЫВАНИЮ, поэтому если avg1 меньше avg2, возвращаем 1 (s1 идет вниз).
        */
        if (avg1 < avg2) return 1;
        if (avg1 > avg2) return -1;
        return 0;
    }

    2. Обертка для сортировки.
    void sort_journal_by_average(Student journal[], int count) {
        /* qsort(массив, количество элементов, размер элемента, функция-компаратор) */
        qsort(journal, count, sizeof(Student), compare_students);
    }

    /* Безопасное чтение строки с поддержкой UTF-8 (кириллицы) */
    void read_string(char *buffer, int max_len) {
        int i = 0;
        int c;
        while ((c = getchar()) != '\n' && c != EOF && i < max_len - 1) {
            buffer[i++] = (char)c;
        }
        buffer[i] = '\0';
        
        /* Если ввели больше, чем влезает в буфер - очищаем остаток строки */
        if (c != '\n' && c != EOF) {
            while ((c = getchar()) != '\n' && c != EOF);
        }
    }

    💡 Почему это сработает:
